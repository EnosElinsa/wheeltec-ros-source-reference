cmake_minimum_required(VERSION 3.16)
project(wheeltec_sim_actuator)
find_package(ament_cmake REQUIRED)
find_package(ignition-gazebo6 REQUIRED)
find_package(ignition-plugin1 REQUIRED COMPONENTS register)
add_library(wheeltec_sim_actuator SHARED src/actuator.cc)
target_compile_features(wheeltec_sim_actuator PUBLIC cxx_std_17)
target_include_directories(wheeltec_sim_actuator PRIVATE include)
target_link_libraries(wheeltec_sim_actuator ignition-gazebo6::ignition-gazebo6 ignition-plugin1::register)
install(TARGETS wheeltec_sim_actuator LIBRARY DESTINATION lib)
if(BUILD_TESTING)
  enable_testing()
  add_executable(actuator_kinematics_test tests/kinematics_test.cc)
  target_compile_features(actuator_kinematics_test PRIVATE cxx_std_17)
  target_include_directories(actuator_kinematics_test PRIVATE include)
  # Keep assertions enabled in release builds: these are executable tests.
  target_compile_options(actuator_kinematics_test PRIVATE -UNDEBUG)
  add_test(NAME actuator_kinematics COMMAND actuator_kinematics_test)
endif()
ament_package()
