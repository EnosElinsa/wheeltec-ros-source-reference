cmake_minimum_required(VERSION 3.8)
project(wheeltec_tf_guard)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(tf2_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
add_executable(tf_authority_guard src/tf_authority_guard.cpp)
target_compile_features(tf_authority_guard PUBLIC cxx_std_14)
target_compile_options(tf_authority_guard PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(tf_authority_guard rclcpp tf2_msgs diagnostic_msgs)
install(TARGETS tf_authority_guard DESTINATION lib/${PROJECT_NAME})
ament_package()
